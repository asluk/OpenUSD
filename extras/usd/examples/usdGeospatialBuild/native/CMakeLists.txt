cmake_minimum_required(VERSION 3.24)
project(GeospatialCandidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
find_package(OpenGL REQUIRED)
find_package(pxr CONFIG REQUIRED)
add_library(candidatePlacement SHARED placement.cpp)
target_compile_definitions(candidatePlacement PRIVATE BUILD_PLACEMENT)
target_include_directories(candidatePlacement PUBLIC ${PXR_INCLUDE_DIRS} ${USD_SDK}/include ${Python3_INCLUDE_DIRS})
target_link_directories(candidatePlacement PUBLIC ${USD_SDK}/lib)
target_link_libraries(candidatePlacement PRIVATE usd usdGeom usdImaging hd gf tf sdf vt plug proj)
add_executable(candidateNative main.cpp)
target_include_directories(candidateNative PRIVATE ${PXR_INCLUDE_DIRS} ${USD_SDK}/include ${Python3_INCLUDE_DIRS})
target_link_directories(candidateNative PRIVATE ${USD_SDK}/lib)
target_link_libraries(candidateNative PRIVATE candidatePlacement usd usdGeom usdImagingGL hd hdx hgi glf garch hio gf tf js sdf vt plug opengl32 user32 gdi32)
add_library(candidateValidation SHARED validation.cpp)
target_include_directories(candidateValidation PRIVATE ${PXR_INCLUDE_DIRS} ${USD_SDK}/include ${Python3_INCLUDE_DIRS})
target_link_directories(candidateValidation PRIVATE ${USD_SDK}/lib)
target_link_libraries(candidateValidation PRIVATE usd usdGeom usdValidation tf sdf vt gf proj)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/placement ${CMAKE_CURRENT_BINARY_DIR}/validation)
configure_file(plugInfo.json plugInfo.json COPYONLY)

# Independent USD reader for candidate discovery, source values and direct origins.
# This target does not choose a model frame or compute resolved scene geometry.
add_executable(proposalScope proposalScope.cpp)
target_include_directories(proposalScope PRIVATE ${PXR_INCLUDE_DIRS} ${USD_SDK}/include ${Python3_INCLUDE_DIRS})
target_link_directories(proposalScope PRIVATE ${USD_SDK}/lib)
target_link_libraries(proposalScope PRIVATE usd usdGeom gf tf sdf vt js proj)

target_link_directories(leansNative PRIVATE ${USD_SDK}/lib)
target_link_libraries(leansNative PRIVATE usd usdGeom gf tf sdf vt js plug proj)
