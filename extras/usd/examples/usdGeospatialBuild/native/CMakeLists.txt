cmake_minimum_required(VERSION 3.24)
project(GeospatialBuildHydra LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenGL REQUIRED)
find_package(pxr CONFIG REQUIRED)
add_executable(geobuildHydra hydraBridge.cpp)
target_include_directories(geobuildHydra PRIVATE ${PXR_INCLUDE_DIRS})
# Only stock USD/Hydra targets. No retired geospatial implementation is linked.
target_link_libraries(geobuildHydra PRIVATE hd hdSt hdx hgi hgiGL glf garch hio gf js tf sdf vt)
if(WIN32)
    target_compile_definitions(geobuildHydra PRIVATE NOMINMAX)
    target_link_libraries(geobuildHydra PRIVATE opengl32 gdi32 user32)
endif()
if(MSVC)
    target_compile_options(geobuildHydra PRIVATE /EHsc /bigobj)
endif()
