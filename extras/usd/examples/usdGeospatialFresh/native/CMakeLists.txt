cmake_minimum_required(VERSION 3.24)
project(GeospatialFresh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(geospatial_fresh reader.cpp)
target_include_directories(geospatial_fresh PRIVATE "${USD_ROOT}/include" "${PROJ_ROOT}/include" "${PYTHON_ROOT}/include")
target_compile_definitions(geospatial_fresh PRIVATE NOMINMAX)
target_link_directories(geospatial_fresh PRIVATE "${USD_ROOT}/lib")
foreach(lib usd_usd usd_usdGeom usd_usdProfiles usd_sdf usd_pcp usd_ar usd_kind usd_gf usd_tf usd_vt usd_plug usd_js usd_trace usd_arch usd_work usd_ts usd_python usd_boost)
  target_link_libraries(geospatial_fresh PRIVATE "${USD_ROOT}/lib/${lib}.lib")
endforeach()
target_link_libraries(geospatial_fresh PRIVATE "${PROJ_ROOT}/lib/proj.lib" "${PYTHON_ROOT}/libs/python310.lib")
